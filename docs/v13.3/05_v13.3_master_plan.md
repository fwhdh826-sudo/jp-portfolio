```python
# backend/engine/regime/hmm_detector.py

from hmmlearn import hmm
import numpy as np

class HMMRegimeDetector:
    """
    隠れマルコフモデルでレジーム判定
    """
    
    def __init__(self):
        self.model = hmm.GaussianHMM(
            n_components=5,  # 5レジーム
            covariance_type="full",
            n_iter=100
        )
        self.is_trained = False
    
    def train(self, historical_features: np.ndarray):
        """
        過去データで学習
        Features: [returns, vix, volume_z, spread, sentiment]
        """
        self.model.fit(historical_features)
        self.is_trained = True
    
    def predict(self, current_features: np.ndarray) -> tuple:
        """
        現在のレジーム + 確率
        """
        if not self.is_trained:
            return "uncertain", 0.0
        
        states = self.model.predict(current_features)
        probs = self.model.predict_proba(current_features)
        
        latest_state = states[-1]
        latest_prob = probs[-1][latest_state]
        
        # 状態をラベルにマッピング（学習結果から）
        regime_label = self._state_to_regime(latest_state)
        
        return regime_label, latest_prob
```

## 8. Multi-Strategy Engine 詳細

### 8.1 4戦略並行実装

```python
# backend/engine/strategies/

class BaseStrategy(ABC):
    @abstractmethod
    def compute_ideal_pf(self, universe, scores, regime) -> dict:
        pass
    
    @abstractmethod
    def expected_metrics(self) -> dict:
        pass

class FrontierStrategy(BaseStrategy):
    """v13.2 の Frontier AI Index"""
    pass

class QualityAdjustedSizeStrategy(BaseStrategy):
    """Asness 2018 - 小型×高クオリティ"""
    
    def compute_ideal_pf(self, universe, scores, regime):
        # Step 1: Quality 4軸（収益性・成長・安全・還元）でスコアリング
        quality_scores = compute_quality_score_4axis(universe, scores)
        
        # Step 2: 小型銘柄だけに絞り込み
        small_caps = [t for t in universe if t.market_cap < 200_000_000_000]
        
        # Step 3: Quality > 60 でJunk除外
        candidates = [
            t for t in small_caps
            if quality_scores[t.ticker] > 60
        ]
        
        # Step 4: 上位30銘柄を等加重
        top30 = sorted(candidates, key=lambda t: quality_scores[t.ticker], reverse=True)[:30]
        weight_each = 1.0 / 30
        
        return {t.ticker: weight_each for t in top30}

class FundamentalWeightedStrategy(BaseStrategy):
    """Arnott 2005 - ファンダメンタル加重"""
    
    def compute_ideal_pf(self, universe, scores, regime):
        # 売上・キャッシュフロー・配当・簿価で加重
        weights = {}
        total_score = 0
        
        for asset in universe:
            score = (
                asset.revenue_5y_avg * 0.30 +
                asset.cf_5y_avg * 0.30 +
                asset.book_value * 0.20 +
                asset.dividend_5y_avg * 0.20
            )
            weights[asset.ticker] = score
            total_score += score
        
        return {k: v / total_score for k, v in weights.items()}

class CrossFactorStrategy(BaseStrategy):
    """Alquist 2018 - サイズ×他ファクター"""
    
    def compute_ideal_pf(self, universe, scores, regime):
        # Cross-Axis Signal が強い銘柄を選出
        cross_signals = {}
        for asset in universe:
            cs = scores[asset.ticker]
            cross_signals[asset.ticker] = (
                cs["size_quality"] * 0.30 +
                cs["size_value"] * 0.25 +
                cs["size_momentum"] * 0.25 +
                cs["quality_value"] * 0.20
            )
        
        # 上位選出
        top = sorted(cross_signals.items(), key=lambda x: x[1], reverse=True)[:25]
        return {ticker: 1.0/25 for ticker, _ in top}
```

```python
class StrategyAggregator:
    def aggregate(
        self,
        strategy_pfs: dict,  # {strategy_name: ideal_pf}
        regime: str,
        method: str = "regime_aware"
    ) -> dict:

        4戦略のPFを統合

        if method == "regime_aware":
            weights = REGIME_STRATEGY_WEIGHTS[regime]
        elif method == "equal":
            weights = {k: 0.25 for k in strategy_pfs.keys()}
        elif method == "sharpe_weighted":
            weights = self.sharpe_weighted_weights(strategy_pfs)
        
        # 各銘柄の最終ウェイト = Σ (戦略重み × 戦略内ウェイト)
        final_pf = {}
        for strat_name, pf in strategy_pfs.items():
            strat_weight = weights[strat_name]
            for ticker, weight in pf.items():
                final_pf[ticker] = final_pf.get(ticker, 0) + strat_weight * weight
        
        return final_pf
```

```python
def path_dependent_rebalance(
    current_pf: dict,
    target_pf: dict,
    historical_30d: dict
) -> list:
    """
    パス依存型リバランス推奨を生成
    """
    recommendations = []
    
    for ticker, target_weight in target_pf.items():
        current_weight = current_pf.get(ticker, 0)
        diff = abs(current_weight - target_weight)
        
        # 資産タイプ判定
        asset_type = classify_asset_type(ticker)
        rules = PATH_DEPENDENT_RULES[asset_type]
        
        # 閾値チェック
        if diff < rules["trigger_threshold"]:
            continue
        
        # 30日相対パフォーマンス
        if rules["consider_30d_relative"]:
            relative_perf = historical_30d[ticker]["relative_to_index"]
            
            if current_weight < target_weight:
                # 買い増し検討
                if relative_perf < rules["buy_low_threshold"]:
                    priority = "high"  # Buy Low
                else:
                    priority = "medium"
            else:
                # 売却検討
                if relative_perf > rules["sell_high_threshold"]:
                    priority = "high"  # Sell High
                else:
                    priority = "medium"
        else:
            priority = "medium"
        
        recommendations.append({
            "ticker": ticker,
            "current": current_weight,
            "target": target_weight,
            "action": "buy" if current_weight < target_weight else "sell",
            "priority": priority,
            "reason": build_reason(ticker, diff, rules)
        })
    
    return sorted(recommendations, key=lambda x: 0 if x["priority"]=="high" else 1)
```
