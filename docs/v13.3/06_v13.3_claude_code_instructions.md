# backend/engine/regime/rule_based.py

def detect_regime_rule_based(market_data: dict) -> str:
    """
    ルールベース即応判定
    """
    vix = market_data["vix"]
    nikkei_5d = market_data["nikkei_5d_return"]
    nikkei_60ma = market_data["nikkei_60ma"]
    nikkei_200ma = market_data["nikkei_200ma"]
    sp500_dd = market_data["sp500_dd_30d"]
    
    if vix > 40 or sp500_dd < -0.20:
        return "crisis"
    if nikkei_60ma < nikkei_200ma and sp500_dd < -0.10:
        return "bear"
    if vix > 25 and nikkei_5d > 0:
        return "bull_volatile"
    if vix < 18 and nikkei_5d >= 0:
        return "bull_calm"
    return "uncertain"

from hmmlearn import hmm
import numpy as np
import joblib

class HMMRegimeDetector:
    def __init__(self, model_path="models/hmm_regime.pkl"):
        self.model_path = model_path
        self.model = hmm.GaussianHMM(
            n_components=5,
            covariance_type="full",
            n_iter=200
        )
        self.is_trained = False
        self._load_if_exists()
    
    def train(self, historical_features: np.ndarray):
        """
        Features: [returns_5d, vix, volume_z, spread, sentiment]
        過去5年以上のデータで学習
        """
        self.model.fit(historical_features)
        self.is_trained = True
        joblib.dump(self.model, self.model_path)
    
    def predict(self, current_features: np.ndarray):
        if not self.is_trained:
            return "uncertain", 0.0
        states = self.model.predict(current_features)
        probs = self.model.predict_proba(current_features)
        return self._state_to_regime(states[-1]), probs[-1].max()
```

import httpx
from datetime import datetime, timedelta

EDINET_BASE = "https://disclosure.edinet-fsa.go.jp/api/v2"

async def fetch_edinet(days_back: int = 1) -> List[NewsItem]:
    """
    EDINET 公式API
    無料、認証不要
    """
    items = []
    today = datetime.now().date()
    
    async with httpx.AsyncClient() as client:
        for d in range(days_back):
            date_str = (today - timedelta(days=d)).strftime("%Y-%m-%d")
            url = f"{EDINET_BASE}/documents.json?date={date_str}&type=2"

                    item = NewsItem(

# backend/engine/scoring/value_score.py

from dataclasses import dataclass
from typing import Dict

@dataclass
class ScoreComponent:
    name: str
    weight: float
    raw_value: float
    normalized: float
    description: str

class ValueScorer:
    AXIS_ID = "value"
    AXIS_NAME_JA = "バリュー"
    
    COMPONENT_WEIGHTS = {
        "per_score": 0.30,
        "pbr_score": 0.25,
        "peg_score": 0.20,
        "div_yield": 0.15,
        "ev_ebitda": 0.10
    }
    
    def calculate(self, ticker: str, financial_data: Dict) -> Dict:
        components = []
        for comp_name, weight in self.COMPONENT_WEIGHTS.items():
            raw = self._get_raw(ticker, comp_name, financial_data)
            normalized = self._normalize(comp_name, raw)
            components.append(ScoreComponent(
                name=comp_name,
                weight=weight,
                raw_value=raw,
                normalized=normalized,
                description=self._explain(comp_name, raw)
            ))
        
        total_score = sum(c.normalized * c.weight for c in components)
        
        return {
            "axis": self.AXIS_ID,
            "name_ja": self.AXIS_NAME_JA,
            "total": round(total_score),
            "components": [c.__dict__ for c in components],
            "explanation": self._build_explanation(components, total_score)
        }
    
    def _normalize(self, comp_name: str, raw: float) -> float:
        """業界基準で 0-100 正規化"""
        norm_table = {
            "per_score": lambda x: max(0, min(100, 100 - (x - 10) * 5)),  # PER 10で50点
            "pbr_score": lambda x: max(0, min(100, 100 - (x - 1) * 30)),  # PBR 1で50点
            "peg_score": lambda x: max(0, min(100, 100 - (x - 1) * 50)),
            "div_yield": lambda x: max(0, min(100, x * 25)),  # 4%で100点
            "ev_ebitda": lambda x: max(0, min(100, 100 - (x - 8) * 10))
        }
        return norm_table[comp_name](raw)
```

### Card 11-1: Path-Dependent Rebalance

```python
# backend/engine/rebalance/path_dependent.py

PATH_DEPENDENT_RULES = {
    "core": {
        "trigger_threshold": 0.05,
        "buy_low_threshold": -0.10,
        "sell_high_threshold": +0.20
    },
    "satellite": {
        "trigger_threshold": 0.15,
        "buy_low_threshold": -0.15,
        "sell_high_threshold": +0.30
    },
    "tactical": {
        "trigger_threshold": 0.25,
        "max_holding_days": 60
    }
}

def path_dependent_rebalance(current_pf, target_pf, historical_30d):
    recommendations = []
    
    for ticker, target_weight in target_pf.items():
        current_weight = current_pf.get(ticker, 0)
        diff = abs(current_weight - target_weight)
        asset_type = classify_asset_type(ticker)
        rules = PATH_DEPENDENT_RULES[asset_type]
        
        if diff < rules["trigger_threshold"]:
            continue
        
        relative_perf = historical_30d.get(ticker, {}).get("relative_to_index", 0)
        
        if current_weight < target_weight:
            if relative_perf < rules["buy_low_threshold"]:
                priority = "high"
                action = "buy_low"
            else:
                priority = "medium"
                action = "buy"
        else:
            if relative_perf > rules["sell_high_threshold"]:
                priority = "high"
                action = "sell_high"
            else:
                priority = "medium"
                action = "sell"
        
        recommendations.append({
            "ticker": ticker,
            "current": current_weight,
            "target": target_weight,
            "action": action,
            "priority": priority,
            "relative_perf_30d": relative_perf
        })
    
    return sorted(recommendations, key=lambda x: 0 if x["priority"]=="high" else 1)
```

# backend/engine/scenarios/yaml_loader.py

import yaml
from pathlib import Path

class ScenarioLoader:
    def __init__(self, yaml_path="backend/engine/scenarios/pre_commitments.yaml"):
        self.scenarios = self.load(yaml_path)
    
    def load(self, path):
        with open(path) as f:
            return yaml.safe_load(f)["scenarios"]
    
    def check_triggers(self, market_state):
        triggered = []
        for scenario in self.scenarios:
            if self.evaluate_trigger(scenario["trigger"], market_state):
                triggered.append(scenario)
        return triggered
    
    def evaluate_trigger(self, trigger, state):
        condition = trigger.get("condition")
        if condition:
            # eval safely
            return safe_eval(condition, state)
        
        conditions_all = trigger.get("conditions_all", [])
        if conditions_all:
            return all(safe_eval(c, state) for c in conditions_all)
        
        return False
    
    def execute_pre_commitment(self, scenario, current_pf):
        """
        事前コミットメントを実行
        """
        actions = scenario["pre_commitment"]
        executed = []
        for action in actions:
            executed.append(self._execute_action(action, current_pf))
        return executed
