```python
class FrontierStrategy(BaseStrategy):
    STRATEGY_ID = "frontier"
    
    def compute_ideal_pf(self, universe, scores, regime):

        expected_returns = self._build_expected_returns(
            universe, scores, regime
        )

            expected_return=weights @ expected_returns,
            expected_vol=np.sqrt(weights @ cov @ weights),
            sharpe_ratio=...,
```

```python
class StrategyAggregator:
    def aggregate(
        self,
        strategy_outputs: dict,
        regime: str,
        method: str = "regime_aware"
    ) -> dict:
        # Step 1: レジーム別重み取得
        weights = REGIME_STRATEGY_WEIGHTS[regime]
        
        # Step 2: 戦略間相関分析
        correlations = self._calc_correlations(strategy_outputs)
        max_corr = max(correlations.values())
        
        if max_corr > 0.7:
            log_warning(f"戦略間相関最大{max_corr:.2f}、分散効果薄")
        
        # Step 3: 統合
        final_pf = {}
        for strat_id, output in strategy_outputs.items():
            strat_weight = weights[strat_id]
            for ticker, weight in output.ideal_pf.items():
                final_pf[ticker] = final_pf.get(ticker, 0) + strat_weight * weight
        
        # Step 4: 期待メトリクス統合
        expected_return = sum(
            weights[sid] * out.expected_return
            for sid, out in strategy_outputs.items()
        )
        
        return {
            "ideal_pf": final_pf,
            "expected_return": expected_return,
            "strategy_correlations": correlations,
            "weights_used": weights,
            "regime": regime,
            "diversification_score": 1 - max_corr  # 相関が低いほど高評価
        }
```

```python
def build_expected_returns(
    universe, market_intel, six_axis_scores, cross_axis_signals, regime
) -> np.ndarray:
    n = len(universe)
    expected_returns = np.zeros(n)
    
    axis_weights = REGIME_AXIS_WEIGHTS[regime]
    
    for i, asset in enumerate(universe):
        # ベースライン
        mu_hist = asset.mean_return_3y
        
        # 6軸調整（レジーム別重み適用）
        scores = six_axis_scores.get(asset.ticker, {})
        alpha_score = sum(
            (scores.get(axis, {}).get("total", 50) - 50) / 50 * 0.001 * weight
            for axis, weight in axis_weights.items()
        )
        
        # Cross-Axis 調整
        cs = cross_axis_signals.get(asset.ticker, {})
        alpha_cross = (
            cs.get("size_quality", 0) / 100 * 0.005 +
            cs.get("anti_junk", 0) / 100 * 0.003
        )
        
        # Size Premium（Alquist 2018）
        size_premium = {
            "small_cap": 0.012,
            "mid_cap": 0.005,
            "large_cap": 0.000
        }[asset.size_segment]
        
        # Market調整
        sentiment = market_intel["sentiment"]["score"]
        keywords = [k["tag"] for k in market_intel.get("keywords", [])]
        
        alpha_market = 0.0
        if sentiment > 70 and asset.is_risk_on:
            alpha_market += 0.005
        elif sentiment < 30:
            alpha_market += 0.003 if asset.is_defensive else -0.003
        
        if "資源高" in keywords and asset.is_energy:
            alpha_market += 0.004
        if "円安" in keywords and asset.is_overseas:
            alpha_market += 0.002
        
        expected_returns[i] = mu_hist + alpha_score + alpha_cross + size_premium + alpha_market
    
    return expected_returns
```

### 6.2 最適化 + Tier A 統合

```python
def optimize_portfolio(
    expected_returns, cov_matrix, asset_meta, regime, risk_aversion=3.0
):
    n = len(expected_returns)
    
    def neg_utility(w):
        ret = w @ expected_returns
        var = w @ cov_matrix @ w
        penalty = calc_soft_penalty(w, asset_meta)
        v3_penalty = calc_v3_diff_penalty(w, asset_meta)
        return -(ret - 0.5 * risk_aversion * var - penalty - v3_penalty)
    
    constraints = [{"type": "eq", "fun": lambda w: np.sum(w) - 1}]
    bounds = [(0, 1) for _ in range(n)]
    w_init = load_v3_initial_weights(asset_meta)
    
    result = minimize(
        neg_utility, w_init,
        method="SLSQP", bounds=bounds, constraints=constraints
    )
    
    optimal_weights = result.x
    optimal_weights = apply_hard_constraints(optimal_weights, asset_meta)
    
    return {
        "weights": optimal_weights,
        "expected_return": optimal_weights @ expected_returns,
        "expected_risk": np.sqrt(optimal_weights @ cov_matrix @ optimal_weights),
        "sharpe_ratio": ...,
        "tier_a_penalty": calc_soft_penalty(optimal_weights, asset_meta),
        "v3_diff": optimal_weights - w_init,
        "regime_used": regime
    }
```

```python
class AdversarialSelfCheck:
    """
    Pro/Anti Thesis を独立に立て、Evaluator が統合
    """
    
    def review(self, frontier_proposal, context):
        # Pro Thesis: 提案を支持する論陣
        pro_thesis = self._build_pro_thesis(frontier_proposal, context)
        
        # Anti Thesis: 提案を否定する論陣（独立LLM呼び出し）
        anti_thesis = self._build_anti_thesis(frontier_proposal, context)
        
        # Evaluator: 両論を比較
        evaluation = self._evaluator(pro_thesis, anti_thesis, context)
        
        return {
            "pro_thesis": pro_thesis,
            "anti_thesis": anti_thesis,
            "evaluation": evaluation,
            "final_verdict": evaluation["verdict"],
            "key_concerns": evaluation["concerns"]
        }
    
    def _build_anti_thesis(self, proposal, context):
        prompt = f"""
        以下の投資提案を「絶対に採用すべきでない」立場から徹底批判してください。
        
        提案: {proposal}
        市況: {context}
        
        観点:
        1. 過去類似シナリオでの失敗
        2. 見落とされたリスク
        3. 集中度・相関リスク
        4. レジーム変化への脆弱性
        5. 機会損失の可能性
        
        強い言葉で反対論を構築してください。
        """
        return call_llm_json(prompt)
```

```python
def vol_target_position_size(
    base_size: float,
    current_vol: float,
    target_vol: float = 0.15,
    max_scale: float = 2.0,
    min_scale: float = 0.5
) -> float:
```

```python
import re

JP_EQUITY_MASTER = load_jp_equity_master()  # 4桁→社名マスタ

def extract_tickers(text: str) -> list:
    """
    記事本文から銘柄コード（4桁数字）を抽出
    """
    # 4桁数字パターン（前後に数字がない）
    pattern = r'(?<![\d])(\d{4})(?![\d])'
    candidates = re.findall(pattern, text)
    
    # マスタで実在チェック
    valid = [c for c in candidates if c in JP_EQUITY_MASTER]
    
    # 重複排除
    return list(set(valid))


def extract_tickers_smart(text: str) -> list:
    """
    高度版: 社名からも抽出
    """
    extracted = extract_tickers(text)
    
    # 社名マッチング
    for ticker, company_info in JP_EQUITY_MASTER.items():
        for alias in company_info["aliases"]:
            if alias in text:
                extracted.append(ticker)
                break
    
    return list(set(extracted))
```

### 10.3 重要度算出

```python
def calc_importance(item: NewsItem, market_intel: dict, holdings: list) -> float:
    score = 50  # ベースライン
    
    # ソース信頼度
    source_weights = {
        "bloomberg": 15,
        "reuters": 15,
        "edinet": 12,
        "tdnet": 10,
        "jpx": 8,
        "yahoo_finance_jp": 7,
        "minkabu": 5,
        "shikiho_online": 8
    }
    score += source_weights.get(item.source_id, 0)
    
    # キーワード一致
    for keyword in market_intel.get("active_keywords", []):
        if keyword in item.title:
            score += 8
    
    # 関連銘柄数
    score += min(20, len(item.related_tickers) * 4)
    
    # 保有銘柄に関連
    if any(t in holdings for t in item.related_tickers):
        score += 15
    
    # 時間減衰（24時間で半減）
    hours_old = (datetime.now() - item.published_at).total_seconds() / 3600
    score -= min(40, hours_old * 1.5)
    
    return max(0, min(100, score))
```
